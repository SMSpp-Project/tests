#
# A "meta"-BlockSolverConfig: a map from a Block classname() to the
# BlockSolverConfig that says how to solve each inner sub-Block of the
# Lagrangian Dual. It is the value of the LagrangianDualSolver
# str_LagBF_BSCfg parameter (set in PPHCfg.txt), and
# LagrangianDualSolver::default_BSCfg_for() dispatches it by classname()
# over the inner Block of each LagBFunction:
#
#   ThermalUnitBlock  -> TUBSCfg-CLP.txt (continuous relaxation of the unit)
#   NuclearUnitBlock  -> TUBSCfg-CLP.txt (the same, for the nuclear unit)
#   *  (catch-all)    -> LPBSCfg.txt     (every other "hard" inner Block)
#
# The "*" entry is the default applied to any inner Block whose classname()
# is not listed explicitly; the match is on the exact classname(), hence a
# NuclearUnitBlock, although a ThermalUnitBlock, is listed by its own name.
# Solving the inner subproblems in their continuous relaxation makes the
# Lagrangian Dual bound coincide with the continuous relaxation of the
# :MILPSolver on the whole UCBlock.
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

# a "meta"-BlockConfig , i.e., a map classname --> its BlockConfig
SimpleConfiguration<std::map<std::string,Configuration*>>

3 # the number of elements in the map

ThermalUnitBlock        *TUBSCfg-CLP.txt
NuclearUnitBlock        *TUBSCfg-CLP.txt
*                       *LPBSCfg.txt
