# The AC counterpart of InnerBSCfg.txt: the same "meta"-BlockSolverConfig
# (a map from a Block classname() to the BlockSolverConfig that solves each
# inner sub-Block of the Lagrangian Dual), but for AC instances. There the
# reactive-power / voltage terms make the inner Blocks (ThermalUnitBlock and
# the ACNetworkBlock that the catch-all solves) QCPs, which HiGHS cannot
# solve, so both are routed to the GUROBI configs TUBSCfg-QCP.txt /
# LPBSCfg-QCP.txt instead of the HiGHS TUBSCfg-CLP.txt / LPBSCfg.txt.
#
#   ThermalUnitBlock  -> TUBSCfg-QCP.txt (GUROBI, QCP-capable)
#   NuclearUnitBlock  -> TUBSCfg-QCP.txt (the same, for the nuclear unit)
#   *  (catch-all)    -> LPBSCfg-QCP.txt (GUROBI, QCP-capable)
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

# a "meta"-BlockConfig , i.e., a map classname --> its BlockConfig
SimpleConfiguration<std::map<std::string,Configuration*>>

3 # the number of elements in the map

ThermalUnitBlock        *TUBSCfg-QCP.txt
NuclearUnitBlock        *TUBSCfg-QCP.txt
*                       *LPBSCfg-QCP.txt
