# A "meta"-BlockSolverConfig: a map from a Block classname() to the
# BlockSolverConfig that says how to solve each inner sub-Block of the
# Lagrangian Dual. It is the value of the LagrangianDualSolver
# str_LagBF_BSCfg parameter (set in LDCfg-DP.txt), and
# LagrangianDualSolver::default_BSCfg_for() dispatches it by classname()
# over the inner Block of each LagBFunction:
#
#   ThermalUnitBlock     -> TUBSCfg-DP.txt  (the unit solved exactly by
#                                            dynamic programming)
#   ConversionUnitBlock  -> CUBSCfg-MIP.txt (the unit solved with its
#                                            integer commitment)
#   *  (catch-all)       -> LPBSCfg.txt     (every other "hard" inner Block)
#
# The "*" entry is the default applied to any inner Block whose classname()
# is not listed explicitly. The units are solved with their integer
# commitment, so that the bound of the Lagrangian Dual is the Lagrangian one
# of the integer problem and the binary variables are binary for the
# proximal penalty of the PrimalProximalHeur [see BSPar-DP.txt].
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

# a "meta"-BlockConfig , i.e., a map classname --> its BlockConfig
SimpleConfiguration<std::map<std::string,Configuration*>>

3 # the number of elements in the map

ThermalUnitBlock        *TUBSCfg-DP.txt
ConversionUnitBlock     *CUBSCfg-MIP.txt
*                       *LPBSCfg.txt
