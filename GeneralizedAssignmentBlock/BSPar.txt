#
# BlockSolverConfig for a GeneralizedAssignmentBlock, registering the six
# Solver to be cross-checked:
#
#   - :MILPSolver            the abstract representation of the whole tree
#                            (MILPCfg.txt)
#   - LagrangianDualSolver   the constraints linking the BinaryKnapsackBlock
#                            relaxed, the Lagrangian dual solved by a
#                            BundleSolver (LDCfg.txt)
#   - LagrangianDualSolver   the same Lagrangian dual, solved by a
#                            SubGradSolver (LDCfg-SG.txt)
#   - RepairGeneralizedAssignmentSolver
#                            the repair heuristic, guided by the point the
#                            Variables hold (RepairCfg.txt)
#   - PrimalProximalHeur     the proximal heuristic on the same Lagrangian
#                            dual, the repair heuristic as its primal
#                            recovery (PPHCfg.txt)
#   - BranchAndXSolver       the Lagrangian dual of the nodes, the same
#                            heuristic at each of them (BXCfg.txt)
#
# The problem being an integer one, the Lagrangian dual is a relaxation of it
# and the battery declares it so (-R ,r,r).
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

BlockSolverConfig     # exact type of the Configuration object

1  # the BlockSolverConfig is a "differential" one

6  # number of (the names of) Solver in this BlockSolverConfig

# now all the names of the Solver - - - - - - - - - - - - - - - - - - - - - -
GRBMILPSolver         # name of 1st Solver
#CPXMILPSolver        #   (uncomment the available backend)
#SCIPMILPSolver
#HiGHSMILPSolver
LagrangianDualSolver  # name of 2nd Solver
LagrangianDualSolver  # name of 3rd Solver
RepairGeneralizedAssignmentSolver  # name of 4th Solver
PrimalProximalHeur    # name of 5th Solver
BranchAndXSolver      # name of 6th Solver

6  # number of ComputeConfig in this BlockSolverConfig

# now all the ComputeConfig - - - - - - - - - - - - - - - - - - - - - - - - -
*MILPCfg.txt          # ComputeConfig of the :MILPSolver
*LDCfg.txt            # ComputeConfig of the LagrangianDualSolver
*LDCfg-SG.txt         # ComputeConfig of the LagrangianDualSolver with a
                      # SubGradSolver inside
*RepairCfg.txt        # ComputeConfig of the
                      # RepairGeneralizedAssignmentSolver
*PPHCfg.txt           # ComputeConfig of the PrimalProximalHeur
*BXCfg.txt            # ComputeConfig of the BranchAndXSolver
