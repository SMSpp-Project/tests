#
# A ComputeConfig for the LagrangianDualSolver relaxing the rows of the
# father of a SATBlock with a structure [see BPar-R.txt and BPar-D.txt],
# each sub-SATBlock being solved by OLL [see LPBSCfg.txt]; it basically is
# the ComputeConfig of the inner Solver, which is a BundleSolver. Its value
# is a lower bound on the optimum of the weighted MaxSAT, which BSPar-LD.txt
# declares to the cross-check.
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

ComputeConfig  # exact type of the ComputeConfig object

1  # f_diff == 0 ==> all non-provided parameters are set to the default value
   # f_diff == 1 ==> all non-provided parameters are not changed

19 # number of integer parameters

# now all the integer parameters
intMPStbl          0  # stabilization: 0 proximal, 1 level, 2 doubly
# first those of LagrangianDualSolver proper
int_LDSlv_iBCopy   0  # the sub-Block are moved, not R3B-copied

# then those of BundleSolver as its inner Solver
intDoEasy          0  # no easy components, not even the LagBFunction
intMaxIter     10000  # max number of iterations for each call
intLogVerb         0  # log verbosity of the main bundle algorithm
intWZNorm          2  # norm of the norm-based stopping condition
intBPar1          20  # items discarded after this many useless iterations
intBPar2         400  # max bundle size per component
intBPar3           1  # max items fetched from Fi() at each iteration
intBPar4           1  # min items fetched from Fi() at each iteration
intBPar6           0  # 2nd parameter of the dynamic max items per iteration
intBPar7          11  # how the global pools are dealt with
intMnSSC           0  # min consecutive SS with the same t to increase t
intMnNSC           1  # min consecutive NS with the same t to decrease t
inttSPar1         12  # long-term t-strategy: 12 = balancing
intMaxThread       1  # the sub-SATBlock one after the other
intMaxNrEvls       2  # max function evaluations at each iteration
intFrcLstSS        0  # all the components computed at the optimum or not
intTrgtMng         0  # bit-wise management of targets and accuracies

16 # number of double parameters

# now all the double parameters
dblRelAcc       1e-6  # relative accuracy required to the solution
dblNZEps        1e-8  # threshold to declare 0 the || residual ||
dbltStar        1e+3  # times || residual || to estimate the gap
dblBPar5           4  # 1st parameter of the dynamic max items per iteration
dblm1          -0.01  # NS if Fi( Lambda1 ) >= Fi( Lambda ) + | m1 | v*
dblm2           0.99  # SS if Fi( Lambda1 ) <= Fi( Lambda ) + ( 1 - m2 ) v*
dblm3           0.99  # NR if \sigma^* < - t * m3 * || z* ||
dblmxIncr         10  # max increase of t
dblmnIncr        1.5  # min increase of t, each time it is increased
dblmxDecr        0.1  # max decrease of t
dblmnDecr       0.66  # min decrease of t, each time it is decreased
dbltMaior       1e+6  # max value of t
dbltMinor       1e-4  # min value of t
dbltInit        1e-2  # initial value of t
dbltSPar2      0.001  # parameter of the long-term t-strategy
dbltSPar3          0  # parameter of the small heuristic changes of t

3  # number of string parameters

# now all the string parameters
# first those of LagrangianDualSolver proper
str_LDSlv_ISName  BundleSolver  # the inner Solver
strMPBSolverCfg   MPBCfg.txt    # BlockSolverConfig of the master problem
str_LagBF_BSCfg   LPBSCfg.txt   # BlockSolverConfig of the sub-SATBlock

0  # number of vector-of-int parameters

# now all the vector-of-int parameters
# [none]

0  # number of vector-of-double parameters

# now all the vector-of-double parameters
# [none]

0  # number of vector-of-string parameters

# now all the vector-of-string parameters
# [none]

# pointer to the "extra" Configuration
*  # [none]

# end of ComputeConfig
