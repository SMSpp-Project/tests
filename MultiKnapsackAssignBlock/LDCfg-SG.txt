#
# ComputeConfig of the LagrangianDualSolver relaxing the constraints that
# link the BinaryKnapsackBlock sub-Block of a MultiKnapsackAssignBlock with a
# SubGradSolver as the inner Solver, i.e., the deflected subgradient method
# of the Volume algorithm; the parameters of LagrangianDualSolver proper are
# those of LDCfg.txt, which has a BundleSolver as the inner Solver. The
# sub-Block get their Solver by classname() from the meta-BlockSolverConfig
# SubBSPar.txt.
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

ComputeConfig         # exact type of the ComputeConfig object

1  # f_diff == 0 ==> all non-provided parameters are set to the default value
   # f_diff == 1 ==> all non-provided parameters are not changed

5  # number of integer parameters

# now all the integer parameters
# first those of LagrangianDualSolver proper
int_LDSlv_iBCopy       0 # 1 if sub-Block must be R3B-copied
int_LDSlv_NNMult       1 # the relaxed constraints are inequalities, hence
                         # their multipliers are nonnegative
# then those of SubGradSolver as its inner Solver
intMaxIter          2000 # MaxIter, max number of iterations for each call
intLogVerb             0 # LogVerb, log verbosity of the subgradient method
intProjection          1 # the subgradient is projected on the tangent cone
                         # of the box of the multipliers

2  # number of double parameters

# now all the double parameters
dblRelAcc           1e-6 # relative accuracy required to solution
dbltStar            1e+2 # stopping parameter: multiplied to || direction ||
                         # to estimate the gap

4  # number of string parameters

# now all the string parameters
# first those of LagrangianDualSolver proper
str_LDSlv_ISName   SubGradSolver  # the inner Solver of LagrangianDualSolver
str_LagBF_BSCfg    SubBSPar.txt   # the (meta) BlockSolverConfig of the
                                  # sub-Block
# then those of SubGradSolver as its inner Solver
strStepsize        ColorTV        # the stepsize rule
strDeflection      Volume         # the deflection rule

0  # number of vector-of-int parameters

# now all the vector-of-int parameters
# [none]

0  # number of vector-of-double parameters

# now all the vector-of-double parameters
# [none]

0  # number of vector-of-string parameters

# now all the vector-of-string parameters
# [none]

# pointer to the "extra" Configuration
*  # [none]
