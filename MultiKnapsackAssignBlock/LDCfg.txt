#
# ComputeConfig of the LagrangianDualSolver relaxing the constraints that
# link the BinaryKnapsackBlock sub-Block of a MultiKnapsackAssignBlock; it
# basically is the ComputeConfig of the inner Solver, which is a
# BundleSolver. The sub-Block get their Solver by classname() from the
# meta-BlockSolverConfig SubBSPar.txt.
#
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
# - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

ComputeConfig         # exact type of the ComputeConfig object

1  # f_diff == 0 ==> all non-provided parameters are set to the default value
   # f_diff == 1 ==> all non-provided parameters are not changed

20 # number of integer parameters

# now all the integer parameters
# first those of LagrangianDualSolver proper
int_LDSlv_iBCopy       0 # 1 if sub-Block must be R3B-copied
int_LDSlv_NNMult       1 # the relaxed constraints are inequalities, hence
                         # their multipliers are nonnegative
# then those of BundleSolver as its inner Solver
intMPStbl              0 # stabilization: 0 = proximal, 1 = level,
                         # 2 = doubly stabilized
intDoEasy              0 # do not do easy components even for LagBFunction
intMaxIter          1000 # MaxIter, max number of iterations for each call
intLogVerb             0 # LogVerb, log verbosity of main Bundle algorithm
intWZNorm              2 # which norm to use in the norm-based stopping
                         # condition
intBPar1              20 # discard items when they have been useless for
                         # <this> iterations
intBPar2             400 # max bundle size per component
intBPar3               1 # max n. of items to fetch from Fi() at each
                         # iteration
intBPar4               1 # min n. of items to fetch from Fi() at each
                         # iteration
intBPar6               0 # second parameter for dynamic max n. of items per
                         # iteration
intBPar7              11 # how to deal with the global pools
intMnSSC               0 # min number of consecutive SS with the same t for
                         # a t increase
intMnNSC               1 # min number of consecutive NS with the same t for
                         # a t decrease
inttSPar1             12 # long-term t-strategy (0 = none, 4 = soft,
                         # 8 = hard, 12 = balancing)
intMaxThread           1
intMaxNrEvls           2 # maximum number of function evaluations at each
                         # iteration
intFrcLstSS            0 # if all the components need be computed at the
                         # optimum
intTrgtMng             2 # bit-wise encoding targets and accuracies
                         # management: bit 1 = upper/lower target set to
                         # each convex/concave component

16 # number of double parameters

# now all the double parameters
dblRelAcc           1e-7 # relative accuracy required to solution
dblNZEps            1e-8 # stopping parameter: threshold to declare 0 the
                         # || residual ||
dbltStar            1e+2 # stopping parameter: multiplied to || residual ||
                         # to estimate the gap
dblBPar5               4 # first parameter for dynamic max n. of items per
                         # iteration
dblm1              -0.01 # a NS is possible if (~)
                         # Fi( Lambda1 ) >= Fi( Lambda ) + | m1 | v*
dblm2               0.99 # a SS is possible if
                         # Fi( Lambda1 ) <= Fi( Lambda ) + ( 1 - m2 ) v*
dblm3               0.99 # a NR is computed if \sigma^* < - t * m3 * || z* ||
dblmxIncr             10 # max increase of t
dblmnIncr            1.5 # min increase of t (each time it is increased)
dblmxDecr            0.1 # max decrease of t
dblmnDecr           0.66 # min decrease of t (each time it is decreased)
dbltMaior           1e+6 # maximum value for t
dbltMinor           1e-4 # minimum value for t
dbltInit            1e-2 # initial value for t
dbltSPar2          0.001 # parameter for the long-term t-strategy
dbltSPar3              0 # parameter for small heuristic t adjustments

3  # number of string parameters

# now all the string parameters
# first those of LagrangianDualSolver proper
str_LDSlv_ISName   BundleSolver  # the inner Solver of LagrangianDualSolver
str_LagBF_BSCfg    SubBSPar.txt  # the (meta) BlockSolverConfig of the
                                 # sub-Block
# then those of BundleSolver as its inner Solver
strMPBSolverCfg    MPBCfg.txt    # BlockSolverConfig of the Solver of the
                                 # MasterProblemBlock

0  # number of vector-of-int parameters

# now all the vector-of-int parameters
# [none]

0  # number of vector-of-double parameters

# now all the vector-of-double parameters
# [none]

0  # number of vector-of-string parameters

# now all the vector-of-string parameters
# [none]

# pointer to the "extra" Configuration
*  # [none]
