   # f_diff == 1 ==> all non-provided parameters are not changed

21 # number of integer parameters

# now all the integer parameters
# stabilization: 0 = proximal, 1 = level, 2 = doubly stabilized
intMPStbl              0
# first those of LagrangianDualSolver proper
int_LDSlv_iBCopy       0 # 1 if sub-Block must be R3B-copied
int_LDSlv_NNMult       0 # the consensus constraints are equalities, hence
                         # their multipliers are free
intPushCostToOwner     1 # the value of intPushCostToOwner for the LagBF

# then those of BundleSolver as its inner Solver
intDoEasy              0 # do not do easy components even for LagBFunction
intMaxIter         10000 # MaxIter, max number of iterations for each call
intLogVerb             0 # LogVerb, log verbosity of main Bundle algorithm

# which norm to use in the norm-based stopping condition
intWZNorm              2
# discard items when they have been useless for <this> iterations
intBPar1              20
intBPar2             400 # max bundle size per component
# max n. of items to fetch from Fi() at each iteration
intBPar3               1
# min n. of items to fetch from Fi() at each iteration
intBPar4               1
# second parameter for dynamic max n. of items per iteration
intBPar6               0
intBPar7              11 # how to deal with the global pools
# min number of consecutive SS with the same t for a t increase
intMnSSC               0
# min number of consecutive NS with the same t for a t decrease
intMnNSC               1
# long-term t-strategy (0 = none, 4 = soft, 8 = hard, 12 = balancing)
inttSPar1             12
intMaxThread           1

# maximum number of function evaluations at each iteration
intMaxNrEvls           2

# if all the components need be computed at the optimum
intFrcLstSS            0

# bit-wise encoding targets and accuracies management:
intTrgtMng             0
                         # bit 0: if lower/upper target is set to each
                         # convex/concave component
                         # bit 1: if upper/lower target is set to each
                         # convex/concave component
                         # bit 2-3: if accuracy is set (2 = upper - lower, 3
                         # = using EpsU)
                         # bit 4: if Lipschitz constant is used to compute
                         # the upper model

16 # number of double parameters

# now all the double parameters
dblRelAcc           1e-7 # relative accuracy required to solution
# stopping parameter: threshold to declare 0 the || residual ||
dblNZEps            1e-8
# stopping parameter: multiplied to || residual || to
dbltStar            1e+3
                         # estimate the gap. The multipliers of the consensus
                         # constraints price the components of the weight
                         # vector, whose scale grows with the number of
                         # chunks:
                         # with the customary 1e+1 the estimate is optimistic
                         # and the bundle stops with a gap of the order of
                         # 1e-4, while this closes it to 1e-7 at no cost in
                         # time, the criterion only becoming stricter
# first parameter for dynamic max n. of items per iteration
dblBPar5               4
# a NS is possible if (~) Fi( Lambda1 ) >= Fi( Lambda ) + | m1 | v*
dblm1              -0.01
# a SS is possible if Fi( Lambda1 ) <= Fi( Lambda ) + ( 1 - m2 ) v*
dblm2               0.99
dblm3               0.99 # a NR is computed if \sigma^* < - t * m3 * || z* ||
dblmxIncr             10 # max increase of t
dblmnIncr            1.5 # min increase of t (each time it is increased)
dblmxDecr            0.1 # max decrease of t
dblmnDecr           0.66 # min decrease of t (each time it is decreased)
dbltMaior           1e+6 # maximum value for t
dbltMinor           1e-4 # minimum value for t
dbltInit            1e-2 # initial value for t
dbltSPar2          0.001 # parameter for the long-term t-strategy
dbltSPar3              0 # parameter for small heuristic t adjustments
